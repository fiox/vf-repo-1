import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
from datetime import datetime, time
import tkinter as tk
from tkinter import filedialog, messagebox, ttk


class ForexBreakoutSystem:
    def __init__(self):
        self.data = None
        self.pip_size = None
        self.breakout_start_time = "00:00"
        self.breakout_end_time = "07:00"
        self.breakout_buffer_pips = 0
        self.stop_loss_pips = 20
        self.take_profit_pips = 30
        self.trades = []
        self.buy_positions = []
        self.sell_positions = []

        # Variabili per controllo prezzo duplicato
        self.last_buy_entry = None
        self.last_sell_entry = None

        # Modalità multiple posizioni
        self.multiple_positions_mode = False

        # Debug stats giornaliere
        self.daily_debug_stats = {}  # key: date → dict with buy_count, sell_count, max_concurrent, buy_pips, sell_pips

    def load_csv_file(self):
        """Carica il file CSV con dialog veloce"""
        try:
            root = tk.Tk()
            root.withdraw()
            root.attributes('-topmost', True)
            file_path = filedialog.askopenfilename(
                title="Seleziona file CSV forex",
                filetypes=[("CSV files", "*.csv"), ("All files", "*.*")]
            )
            root.destroy()
            if not file_path:
                print("Nessun file selezionato")
                return False
            print("Caricamento file in corso...")
            # Leggi CSV velocemente
            self.data = pd.read_csv(file_path, header=None,
                                  names=['Date', 'Time', 'Open', 'High', 'Low', 'Close', 'Volume'])
            # Converti datetime una sola volta
            self.data['DateTime'] = pd.to_datetime(self.data['Date'] + ' ' + self.data['Time'])
            self.data = self.data.sort_values('DateTime').reset_index(drop=True)
            # Estrai solo date e time come stringhe per velocità
            self.data['Date_str'] = self.data['Date']
            self.data['Time_str'] = self.data['Time']
            print(f"✓ Caricati {len(self.data)} record")
            print(f"Periodo: {self.data['DateTime'].iloc[0]} - {self.data['DateTime'].iloc[-1]}")
            return True
        except Exception as e:
            print(f"Errore nel caricamento: {str(e)}")
            return False

    def detect_pip_size(self):
        """Rileva pip size velocemente"""
        if self.data is None:
            return 0.0001
        # Prendi un campione piccolo
        sample = self.data['Close'].head(50)
        # Conta decimali più comune
        decimals = []
        for price in sample:
            if '.' in str(price):
                dec_count = len(str(price).split('.')[-1])
                decimals.append(dec_count)
        if not decimals:
            return 0.0001
        most_common = max(set(decimals), key=decimals.count)
        pip_sizes = {2: 0.01, 3: 0.001, 4: 0.0001, 5: 0.00001}
        return pip_sizes.get(most_common, 0.0001)

    def setup_parameters_console(self):
        """Setup parametri da console per velocità"""
        self.pip_size = self.detect_pip_size()
        print(f"\n=== SETUP PARAMETRI ===")
        print(f"Pip size rilevato: {self.pip_size}")
        # Pip size
        pip_input = input(f"Pip size [{self.pip_size}]: ").strip()
        if pip_input:
            try:
                self.pip_size = float(pip_input)
            except:
                print("Valore non valido, uso quello rilevato")
        # Range orario
        print(f"Range breakout attuale: {self.breakout_start_time} - {self.breakout_end_time}")
        range_input = input("Nuovo range (es: 00:00-07:00) [INVIO per mantenere]: ").strip()
        if range_input and '-' in range_input:
            try:
                start, end = range_input.split('-')
                self.breakout_start_time = start.strip()
                self.breakout_end_time = end.strip()
            except:
                print("Formato non valido, mantengo impostazioni")
        # Altri parametri
        buffer_input = input(f"Pips buffer oltre range [{self.breakout_buffer_pips}]: ").strip()
        if buffer_input:
            try:
                self.breakout_buffer_pips = float(buffer_input)
            except:
                pass
        sl_input = input(f"Stop Loss pips [{self.stop_loss_pips}]: ").strip()
        if sl_input:
            try:
                self.stop_loss_pips = float(sl_input)
            except:
                pass
        tp_input = input(f"Take Profit pips [{self.take_profit_pips}]: ").strip()
        if tp_input:
            try:
                self.take_profit_pips = float(tp_input)
            except:
                pass

        # Scelta della modalità posizioni
        print("\nModalità posizioni:")
        print("1. Una sola posizione alla volta")
        print("2. Più posizioni contemporanee (ad ogni candela che rompe il range)")
        mode_choice = input("Scegli [1/2]: ").strip()
        if mode_choice == '2':
            self.multiple_positions_mode = True
            print("✓ Modalità: multiple posizioni abilitata")
        else:
            self.multiple_positions_mode = False
            print("✓ Modalità: una sola posizione alla volta")

        print(f"\n✓ Parametri confermati:")
        print(f"  - Pip size: {self.pip_size}")
        print(f"  - Range: {self.breakout_start_time} - {self.breakout_end_time}")
        print(f"  - Buffer: {self.breakout_buffer_pips} pips")
        print(f"  - Stop Loss: {self.stop_loss_pips} pips")
        print(f"  - Take Profit: {self.take_profit_pips} pips")
        return True

    def calculate_daily_ranges_fast(self):
        """Calcola range giornalieri velocemente"""
        print("Calcolo range giornalieri...")
        # Converti time strings per confronto
        start_time = self.breakout_start_time
        end_time = self.breakout_end_time
        # Filtra dati nel range orario
        mask = (self.data['Time_str'] >= start_time) & (self.data['Time_str'] <= end_time)
        range_data = self.data[mask].copy()
        if range_data.empty:
            print("⚠️ Nessun dato nel range orario specificato!")
            return {}
        # Raggruppa per data e calcola min/max
        daily_ranges = {}
        for date, group in range_data.groupby('Date_str'):
            daily_ranges[date] = {
                'high': group['High'].max(),
                'low': group['Low'].min()
            }
        print(f"✓ Calcolati {len(daily_ranges)} range giornalieri")
        return daily_ranges

    def run_backtest_fast(self):
        """Backtest ottimizzato"""
        print("Avvio backtest...")
        daily_ranges = self.calculate_daily_ranges_fast()
        if not daily_ranges:
            return
        self.trades = []
        self.buy_positions = []
        self.sell_positions = []
        self.last_buy_entry = None
        self.last_sell_entry = None

        # Pre-calcola alcune variabili
        pip_value = self.pip_size
        buffer_value = self.breakout_buffer_pips * pip_value
        sl_value = self.stop_loss_pips * pip_value
        tp_value = self.take_profit_pips * pip_value
        total_rows = len(self.data)
        print_interval = max(1000, total_rows // 20)  # Progress ogni 5%
        for i in range(total_rows):
            if i % print_interval == 0:
                print(f"Processando... {i/total_rows*100:.1f}%")
            row = self.data.iloc[i]
            current_date = row['Date_str']
            current_time_str = row['Time_str']
            current_high = row['High']
            current_low = row['Low']
            current_close = row['Close']
            current_datetime = row['DateTime']

            # Inizializza il contatore per il giorno corrente
            if current_date not in self.daily_debug_stats:
                self.daily_debug_stats[current_date] = {
                    'buy_count': 0,
                    'sell_count': 0,
                    'max_concurrent': 0,
                    'buy_pips': 0.0,
                    'sell_pips': 0.0
                }

            # Calcola quante posizioni sono aperte in totale
            total_open_positions = len(self.buy_positions) + len(self.sell_positions)
            if total_open_positions > self.daily_debug_stats[current_date]['max_concurrent']:
                self.daily_debug_stats[current_date]['max_concurrent'] = total_open_positions

            # Skip se non abbiamo range per questo giorno
            if current_date not in daily_ranges:
                continue
            # Skip durante il range di setup
            if current_time_str >= self.breakout_start_time and current_time_str <= self.breakout_end_time:
                continue

            range_high = daily_ranges[current_date]['high']
            range_low = daily_ranges[current_date]['low']

            # Chiudi posizioni a fine giornata
            if i < total_rows - 1:
                next_date = self.data.iloc[i + 1]['Date_str']
                if next_date != current_date:
                    self.close_all_positions(current_close, current_datetime, "End of Day")

            # Gestisci posizioni esistenti
            self.manage_existing_positions(current_high, current_low, current_datetime)

            # Cerca nuovi breakout
            breakout_high_level = range_high + self.breakout_buffer_pips * self.pip_size
            breakout_low_level = range_low - self.breakout_buffer_pips * self.pip_size

            # Breakout rialzista - apri BUY
            if current_high > breakout_high_level:
                if self.multiple_positions_mode or not self.buy_positions:
                    entry_price = breakout_high_level
                    if self.last_buy_entry != entry_price:
                        position = {
                            'type': 'BUY',
                            'entry_price': entry_price,
                            'entry_time': current_datetime,
                            'stop_loss': entry_price - sl_value,
                            'take_profit': entry_price + tp_value,
                            'date': current_date
                        }
                        self.buy_positions.append(position)
                        self.last_buy_entry = entry_price
                        self.daily_debug_stats[current_date]['buy_count'] += 1

            # Breakout ribassista - apri SELL
            if current_low < breakout_low_level:
                if self.multiple_positions_mode or not self.sell_positions:
                    entry_price = breakout_low_level
                    if self.last_sell_entry != entry_price:
                        position = {
                            'type': 'SELL',
                            'entry_price': entry_price,
                            'entry_time': current_datetime,
                            'stop_loss': entry_price + sl_value,
                            'take_profit': entry_price - tp_value,
                            'date': current_date
                        }
                        self.sell_positions.append(position)
                        self.last_sell_entry = entry_price
                        self.daily_debug_stats[current_date]['sell_count'] += 1

        # Chiudi posizioni rimaste aperte
        if self.buy_positions or self.sell_positions:
            last_price = self.data.iloc[-1]['Close']
            last_time = self.data.iloc[-1]['DateTime']
            self.close_all_positions(last_price, last_time, "End of Data")
        print(f"✓ Backtest completato! {len(self.trades)} trade eseguiti")

    def manage_existing_positions(self, current_high, current_low, current_time):
        """Gestisce posizioni esistenti"""
        # Gestisci BUY
        for pos in self.buy_positions[:]:
            if current_low <= pos['stop_loss']:
                exit_price = pos['stop_loss']
                pips = (exit_price - pos['entry_price']) / self.pip_size
                self.daily_debug_stats[pos['date']]['buy_pips'] += round(pips, 1)
                self.close_position(pos, exit_price, current_time, "Stop Loss")
                self.buy_positions.remove(pos)
            elif current_high >= pos['take_profit']:
                exit_price = pos['take_profit']
                pips = (exit_price - pos['entry_price']) / self.pip_size
                self.daily_debug_stats[pos['date']]['buy_pips'] += round(pips, 1)
                self.close_position(pos, exit_price, current_time, "Take Profit")
                self.buy_positions.remove(pos)

        # Gestisci SELL
        for pos in self.sell_positions[:]:
            if current_high >= pos['stop_loss']:
                exit_price = pos['stop_loss']
                pips = (pos['entry_price'] - exit_price) / self.pip_size
                self.daily_debug_stats[pos['date']]['sell_pips'] += round(pips, 1)
                self.close_position(pos, exit_price, current_time, "Stop Loss")
                self.sell_positions.remove(pos)
            elif current_low <= pos['take_profit']:
                exit_price = pos['take_profit']
                pips = (pos['entry_price'] - exit_price) / self.pip_size
                self.daily_debug_stats[pos['date']]['sell_pips'] += round(pips, 1)
                self.close_position(pos, exit_price, current_time, "Take Profit")
                self.sell_positions.remove(pos)

    def close_position(self, position, exit_price, exit_time, reason):
        """Chiude una singola posizione"""
        if position['type'] == 'BUY':
            pips = (exit_price - position['entry_price']) / self.pip_size
        else:
            pips = (position['entry_price'] - exit_price) / self.pip_size
        trade = {
            'entry_time': position['entry_time'],
            'exit_time': exit_time,
            'type': position['type'],
            'entry_price': position['entry_price'],
            'exit_price': exit_price,
            'pips': round(pips, 1),
            'reason': reason
        }
        self.trades.append(trade)

    def close_all_positions(self, exit_price, exit_time, reason):
        """Chiude tutte le posizioni aperte"""
        for pos in self.buy_positions:
            self.close_position(pos, exit_price, exit_time, reason)
        for pos in self.sell_positions:
            self.close_position(pos, exit_price, exit_time, reason)
        self.buy_positions.clear()
        self.sell_positions.clear()

    def calculate_statistics(self):
        """Calcola e mostra statistiche"""
        if not self.trades:
            print("Nessun trade eseguito!")
            return None
        winning_trades = [t for t in self.trades if t['pips'] > 0]
        losing_trades = [t for t in self.trades if t['pips'] <= 0]
        total_trades = len(self.trades)
        win_count = len(winning_trades)
        loss_count = len(losing_trades)
        win_rate = (win_count / total_trades) * 100 if total_trades > 0 else 0
        total_pips = sum(t['pips'] for t in self.trades)
        avg_win = np.mean([t['pips'] for t in winning_trades]) if winning_trades else 0
        avg_loss = np.mean([t['pips'] for t in losing_trades]) if losing_trades else 0
        buy_trades = [t for t in self.trades if t['type'] == 'BUY']
        sell_trades = [t for t in self.trades if t['type'] == 'SELL']
        print(f"\n{'='*50}")
        print(f"{'STATISTICHE TRADING':^50}")
        print(f"{'='*50}")
        print(f"Totale trade: {total_trades}")
        print(f"  - BUY: {len(buy_trades)}")
        print(f"  - SELL: {len(sell_trades)}")
        print(f"\nTrade vincenti: {win_count} ({win_rate:.1f}%)")
        print(f"Trade perdenti: {loss_count} ({100-win_rate:.1f}%)")
        print(f"\nPIPS TOTALI: {total_pips:.1f}")
        print(f"Media vincita: {avg_win:.1f} pips")
        print(f"Media perdita: {avg_loss:.1f} pips")
        if avg_loss != 0:
            profit_factor = abs(sum(t['pips'] for t in winning_trades) / sum(t['pips'] for t in losing_trades))
            print(f"Profit Factor: {profit_factor:.2f}")
        return {
            'total_trades': total_trades,
            'win_rate': win_rate,
            'total_pips': total_pips,
            'avg_win': avg_win,
            'avg_loss': avg_loss
        }

    def plot_equity_curve(self):
        """Mostra equity curve"""
        if not self.trades:
            print("Nessun trade da graficare")
            return
        equity = [0]
        dates = []
        cumulative_pips = 0
        for trade in self.trades:
            cumulative_pips += trade['pips']
            equity.append(cumulative_pips)
            dates.append(trade['exit_time'])

        plt.figure(figsize=(14, 8))
        plt.plot(range(len(equity)), equity, 'b-', linewidth=2, label='Equity')
        plt.title('Equity Curve - Pips Cumulativi', fontsize=16, fontweight='bold')
        plt.xlabel('Numero Trade')
        plt.ylabel('Pips')
        plt.grid(True, alpha=0.3)
        plt.axhline(y=0, color='r', linestyle='--', alpha=0.7, label='Break Even')
        plt.fill_between(range(len(equity)), equity, 0,
                        where=np.array(equity) >= 0, color='green', alpha=0.2, label='Profit')
        plt.fill_between(range(len(equity)), equity, 0,
                        where=np.array(equity) < 0, color='red', alpha=0.2, label='Loss')

        stats = self.calculate_statistics()
        if stats:
            textstr = f"Totale Pips: {stats['total_pips']:.1f}\n"
            textstr += f"Win Rate: {stats['win_rate']:.1f}%\n"
            textstr += f"Totale Trade: {stats['total_trades']}\n"
            textstr += f"Avg Win: {stats['avg_win']:.1f}\n"
            textstr += f"Avg Loss: {stats['avg_loss']:.1f}"
            props = dict(boxstyle='round', facecolor='lightblue', alpha=0.8)
            plt.text(0.02, 0.98, textstr, transform=plt.gca().transAxes, fontsize=11,
                     verticalalignment='top', bbox=props, fontweight='bold')
        plt.legend()
        plt.tight_layout()
        plt.show()

    def save_debug_stats_to_csv(self):
        """Salva le statistiche di debug in un file CSV"""
        if not self.daily_debug_stats:
            print("⚠️ Nessuna statistica di debug disponibile")
            return

        try:
            root = tk.Tk()
            root.withdraw()
            root.attributes('-topmost', True)
            file_path = filedialog.asksaveasfilename(
                defaultextension=".csv",
                filetypes=[("CSV files", "*.csv"), ("All files", "*.*")],
                title="Salva file CSV delle statistiche di debug"
            )
            root.destroy()

            if not file_path:
                print("⚠️ Salvataggio annullato")
                return

            with open(file_path, 'w') as f:
                f.write('Data,Buy,Sell,MaxPos,BuyPips,SellPips\n')
                for date, stats in sorted(self.daily_debug_stats.items()):
                    f.write(f"{date},{stats['buy_count']},{stats['sell_count']},{stats['max_concurrent']},{stats['buy_pips']:.1f},{stats['sell_pips']:.1f}\n")
            print(f"✓ Statistiche di debug salvate in: {file_path}")
        except Exception as e:
            print(f"Errore nel salvataggio del file di debug: {str(e)}")

    def save_positions_for_date_to_csv(self):
        """Chiede all'utente una data e salva un CSV con tutte le posizioni aperte/chiusure quel giorno"""
        print("\nVuoi salvare le posizioni di una giornata specifica?")
        date_input = input("Inserisci una data nel formato YYYY-MM-DD (es: 2024-01-15) oppure INVIO per saltare: ").strip()
        if not date_input:
            print("Operazione annullata.")
            return

        try:
            target_date = datetime.strptime(date_input, "%Y-%m-%d").strftime("%Y-%m-%d")
        except ValueError:
            print("Formato data non valido.")
            return

        positions_on_date = [t for t in self.trades if t['entry_time'].strftime("%Y-%m-%d") == target_date]
        if not positions_on_date:
            print(f"Nessuna posizione trovata per la data {target_date}.")
            return

        try:
            root = tk.Tk()
            root.withdraw()
            root.attributes('-topmost', True)
            file_path = filedialog.asksaveasfilename(
                defaultextension=".csv",
                initialfile=f"posizioni_{target_date}.csv",
                filetypes=[("CSV files", "*.csv"), ("All files", "*.*")],
                title=f"Salva posizioni per {target_date}"
            )
            root.destroy()

            if not file_path:
                print("⚠️ Salvataggio annullato")
                return

            with open(file_path, 'w') as f:
                f.write('Data,OraApertura,PrezzoApertura,OraChiusura,PrezzoChiusura,Tipo,Pips\n')
                for trade in positions_on_date:
                    entry_time = trade['entry_time'].strftime("%H:%M")
                    exit_time = trade['exit_time'].strftime("%H:%M")
                    f.write(f"{trade['entry_time'].strftime('%Y-%m-%d')},{entry_time},{trade['entry_price']:.5f},{exit_time},{trade['exit_price']:.5f},{trade['type']},{trade['pips']:.1f}\n")
            print(f"✓ Posizioni salvate in: {file_path}")
        except Exception as e:
            print(f"Errore nel salvataggio: {str(e)}")

    def ask_generate_csv(self):
        """Chiede all'utente se generare i file CSV"""
        print("\nVuoi generare i file CSV di output?")
        generate = input("S/N ").strip().lower()
        if generate == 's':
            print("\nCosa vuoi generare?")
            print("1. Solo il file delle statistiche giornaliere")
            print("2. Solo il file delle posizioni per una data")
            print("3. Entrambi i file")
            choice = input("Scelta [1/2/3]: ").strip()
            if choice in ['1', '3']:
                self.save_debug_stats_to_csv()
            if choice in ['2', '3']:
                self.save_positions_for_date_to_csv()
        else:
            print("Nessun file CSV generato.")

    def run(self):
        """Esegue il sistema completo"""
        print("🚀 SISTEMA BREAKOUT FOREX")
        print("=" * 50)
        # 1. Carica file
        if not self.load_csv_file():
            return
        # 2. Setup parametri
        if not self.setup_parameters_console():
            return
        # 3. Backtest
        self.run_backtest_fast()
        # 4. Risultati
        self.calculate_statistics()
        # 5. Salvataggio CSV
        self.ask_generate_csv()
        # 6. Grafico
        input("\nPremi INVIO per vedere l'equity curve...")
        self.plot_equity_curve()


# Esecuzione
if __name__ == "__main__":
    system = ForexBreakoutSystem()
    system.run()
