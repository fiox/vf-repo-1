#property indicator_chart_window
#property strict

input int DaysBack = 100;        // giorni da controllare
input color TextColor = clrBlack;
input int FontSize = 10;

// Struttura per contare le occorrenze
struct CounterStruct
{
   int m1BullM5Bull;  // M1 bull -> M5 bull
   int m1BullM5Bear;  // M1 bull -> M5 bear
   int m1BearM5Bull;  // M1 bear -> M5 bull
   int m1BearM5Bear;  // M1 bear -> M5 bear
   
   // Nuovi contatori per la prima candela M1
   int firstM1Bull;   // Prima candela M1 bull
   int firstM1Bear;   // Prima candela M1 bear
   int firstM1BullM5Bull;  // Prima candela M1 bull -> M5 bull
   int firstM1BullM5Bear;  // Prima candela M1 bull -> M5 bear
   int firstM1BearM5Bull;  // Prima candela M1 bear -> M5 bull
   int firstM1BearM5Bear;  // Prima candela M1 bear -> M5 bear
};

CounterStruct counter;

//+------------------------------------------------------------------+
int OnInit()
{
   // Pulizia oggetti
   ObjectsDeleteAll(0, OBJ_LABEL);
   return(INIT_SUCCEEDED);
}

//+------------------------------------------------------------------+
void OnDeinit(const int reason)
{
   ObjectsDeleteAll(0, OBJ_LABEL);
}

//+------------------------------------------------------------------+
int OnCalculate(const int rates_total,
                const int prev_calculated,
                const datetime &time[],
                const double &open[],
                const double &high[],
                const double &low[],
                const double &close[],
                const long &tick_volume[],
                const long &volume[],
                const int &spread[])
{
   // Reset contatori
   counter.m1BullM5Bull = 0;
   counter.m1BullM5Bear = 0;
   counter.m1BearM5Bull = 0;
   counter.m1BearM5Bear = 0;
   
   // Reset nuovi contatori
   counter.firstM1Bull = 0;
   counter.firstM1Bear = 0;
   counter.firstM1BullM5Bull = 0;
   counter.firstM1BullM5Bear = 0;
   counter.firstM1BearM5Bull = 0;
   counter.firstM1BearM5Bear = 0;
   
   // Controllo che sia timeframe M5
   if(Period() != PERIOD_M5)
   {
      ObjectsDeleteAll(0, OBJ_LABEL);
      CreateLabel("ErrorLabel", "Questo indicatore funziona solo su timeframe M5", 10, 20, TextColor, FontSize);
      return(rates_total);
   }
   
   datetime currentTime = TimeCurrent();
   datetime startTime = currentTime - DaysBack * 86400;
   
   int totalBars = rates_total;
   int startBar = totalBars - 1;
   
   // Troviamo la prima barra nel nostro periodo di analisi
   for(int i = 0; i < totalBars; i++)
   {
      if(time[i] >= startTime)
      {
         startBar = i;
         break;
      }
   }
   
   // Per ogni candela M5
   for(int i = startBar; i < totalBars; i++)
   {
      datetime m5Time = time[i];
      bool m5Bull = (close[i] > open[i]);
      
      // Calcola il tempo corrispondente al minuto esatto (xx:00, xx:05, xx:10...)
      datetime exactTime = m5Time;
      
      // Trova la prima candela M1 all'interno di questa M5
      datetime firstM1Time = exactTime;
      double firstM1Open = iOpen(Symbol(), PERIOD_M1, iBarShift(Symbol(), PERIOD_M1, firstM1Time, false));
      double firstM1Close = iClose(Symbol(), PERIOD_M1, iBarShift(Symbol(), PERIOD_M1, firstM1Time, false));
      
      // Controlliamo se abbiamo dati validi per la prima candela M1
      if(firstM1Open == 0 || firstM1Close == 0)
         continue;
         
      bool firstM1Bull = (firstM1Close > firstM1Open);
      
      // Incrementa il contatore appropriato
      if(firstM1Bull && m5Bull)
         counter.m1BullM5Bull++;
      else if(firstM1Bull && !m5Bull)
         counter.m1BullM5Bear++;
      else if(!firstM1Bull && m5Bull)
         counter.m1BearM5Bull++;
      else // !firstM1Bull && !m5Bull
         counter.m1BearM5Bear++;
      
      // Incrementa i nuovi contatori per la prima candela M1
      if(firstM1Bull)
      {
         counter.firstM1Bull++;
         if(m5Bull)
            counter.firstM1BullM5Bull++;
         else
            counter.firstM1BullM5Bear++;
      }
      else
      {
         counter.firstM1Bear++;
         if(m5Bull)
            counter.firstM1BearM5Bull++;
         else
            counter.firstM1BearM5Bear++;
      }
   }
   
   // Calcola i totali per M1 Bull e M1 Bear
   int totalM1Bull = counter.m1BullM5Bull + counter.m1BullM5Bear;
   int totalM1Bear = counter.m1BearM5Bull + counter.m1BearM5Bear;
   int total = totalM1Bull + totalM1Bear;
   
   // Calcola i totali per la prima candela M1
   int totalFirstM1 = counter.firstM1Bull + counter.firstM1Bear;
   
   // Calcola le percentuali per ogni categoria rispetto al proprio gruppo (Bull o Bear)
   string pct1 = totalM1Bull > 0 ? " (" + DoubleToString(100.0 * counter.m1BullM5Bull / totalM1Bull, 1) + "%)" : " (0%)";
   string pct2 = totalM1Bull > 0 ? " (" + DoubleToString(100.0 * counter.m1BullM5Bear / totalM1Bull, 1) + "%)" : " (0%)";
   string pct3 = totalM1Bear > 0 ? " (" + DoubleToString(100.0 * counter.m1BearM5Bull / totalM1Bear, 1) + "%)" : " (0%)";
   string pct4 = totalM1Bear > 0 ? " (" + DoubleToString(100.0 * counter.m1BearM5Bear / totalM1Bear, 1) + "%)" : " (0%)";
   
   // Calcola le percentuali per le nuove categorie
   string pctFirst1 = totalFirstM1 > 0 ? " (" + DoubleToString(100.0 * counter.firstM1Bull / totalFirstM1, 1) + "%)" : " (0%)";
   string pctFirst2 = totalFirstM1 > 0 ? " (" + DoubleToString(100.0 * counter.firstM1Bear / totalFirstM1, 1) + "%)" : " (0%)";
   
   string pctFirst3 = counter.firstM1Bull > 0 ? " (" + DoubleToString(100.0 * counter.firstM1BullM5Bull / counter.firstM1Bull, 1) + "%)" : " (0%)";
   string pctFirst4 = counter.firstM1Bull > 0 ? " (" + DoubleToString(100.0 * counter.firstM1BullM5Bear / counter.firstM1Bull, 1) + "%)" : " (0%)";
   string pctFirst5 = counter.firstM1Bear > 0 ? " (" + DoubleToString(100.0 * counter.firstM1BearM5Bull / counter.firstM1Bear, 1) + "%)" : " (0%)";
   string pctFirst6 = counter.firstM1Bear > 0 ? " (" + DoubleToString(100.0 * counter.firstM1BearM5Bear / counter.firstM1Bear, 1) + "%)" : " (0%)";
   
   // Visualizza i risultati
   ObjectsDeleteAll(0, OBJ_LABEL);
   
   // Intestazioni per i gruppi
   CreateLabel("HeaderBull", "Gruppo M1 Bull: " + IntegerToString(totalM1Bull) + " casi (" + 
               DoubleToString(total > 0 ? 100.0 * totalM1Bull / total : 0, 1) + "% del totale)", 10, 20, TextColor, FontSize);
   
   CreateLabel("Label1", "  M1 Bull, M1 Bull > M5 Bull: " + IntegerToString(counter.m1BullM5Bull) + pct1, 10, 40, TextColor, FontSize);
   CreateLabel("Label2", "  M1 Bull, M1 Bull > M5 Bear: " + IntegerToString(counter.m1BullM5Bear) + pct2, 10, 60, TextColor, FontSize);
   
   CreateLabel("HeaderBear", "Gruppo M1 Bear: " + IntegerToString(totalM1Bear) + " casi (" + 
               DoubleToString(total > 0 ? 100.0 * totalM1Bear / total : 0, 1) + "% del totale)", 10, 80, TextColor, FontSize);
   
   CreateLabel("Label3", "  M1 Bear, M1 Bear > M5 Bull: " + IntegerToString(counter.m1BearM5Bull) + pct3, 10, 100, TextColor, FontSize);
   CreateLabel("Label4", "  M1 Bear, M1 Bear > M5 Bear: " + IntegerToString(counter.m1BearM5Bear) + pct4, 10, 120, TextColor, FontSize);
   
   CreateLabel("Label5", "Totale analizzato: " + IntegerToString(total) + " barre", 10, 140, TextColor, FontSize);
   
   // Aggiungi nuove statistiche sulla prima candela M1
   CreateLabel("HeaderFirstM1", "Solo Prima Candela M1 - Totale: " + IntegerToString(totalFirstM1) + " casi", 10, 170, TextColor, FontSize);
   
   CreateLabel("LabelFirstBull", "  Prima candela M1 Bull: " + IntegerToString(counter.firstM1Bull) + pctFirst1, 10, 190, TextColor, FontSize);
   CreateLabel("LabelFirstBear", "  Prima candela M1 Bear: " + IntegerToString(counter.firstM1Bear) + pctFirst2, 10, 210, TextColor, FontSize);
   
   CreateLabel("HeaderFirstDetails", "Dettagli Prima Candela M1:", 10, 240, TextColor, FontSize);
   
   CreateLabel("LabelFirst1", "  M1 Bull > M5 Bull: " + IntegerToString(counter.firstM1BullM5Bull) + pctFirst3, 10, 260, TextColor, FontSize);
   CreateLabel("LabelFirst2", "  M1 Bull > M5 Bear: " + IntegerToString(counter.firstM1BullM5Bear) + pctFirst4, 10, 280, TextColor, FontSize);
   CreateLabel("LabelFirst3", "  M1 Bear > M5 Bull: " + IntegerToString(counter.firstM1BearM5Bull) + pctFirst5, 10, 300, TextColor, FontSize);
   CreateLabel("LabelFirst4", "  M1 Bear > M5 Bear: " + IntegerToString(counter.firstM1BearM5Bear) + pctFirst6, 10, 320, TextColor, FontSize);
   
   return(rates_total);
}

//+------------------------------------------------------------------+
void CreateLabel(string name, string text, int x, int y, color clr, int fontSize)
{
   ObjectCreate(0, name, OBJ_LABEL, 0, 0, 0);
   ObjectSetInteger(0, name, OBJPROP_CORNER, CORNER_LEFT_UPPER);
   ObjectSetInteger(0, name, OBJPROP_XDISTANCE, x);
   ObjectSetInteger(0, name, OBJPROP_YDISTANCE, y);
   ObjectSetInteger(0, name, OBJPROP_COLOR, clr);
   ObjectSetInteger(0, name, OBJPROP_FONTSIZE, fontSize);
   ObjectSetString(0, name, OBJPROP_TEXT, text);
}
