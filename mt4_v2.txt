#property indicator_style1 STYLE_SOLID
#property indicator_width1 2
#property indicator_color1 clrDarkGreen    // Bullish Trend Strength
#property indicator_style2 STYLE_SOLID
#property indicator_width2 2
#property indicator_color2 clrLime     // Bullish Retracement Strength
#property indicator_style3 STYLE_SOLID
#property indicator_width3 2
#property indicator_color3 clrMaroon      // Bearish Trend Strength
#property indicator_style4 STYLE_SOLID
#property indicator_width4 2
#property indicator_color4 clrRed   // Bearish Retracement Strength

// Input Parameters
extern int TickPerCandle = 1000;           // Ticks for Trend (default 1000)
extern int RetracementUpdateInterval = 250; // Ticks for Retracement (default 250)
extern int LookbackPeriod = 500;           // Lookback Period (default 500)

// Indicator buffers
double BullTrendStrengthBuffer[];
double BullRetracementBuffer[];
double BearTrendStrengthBuffer[];
double BearRetracementBuffer[];

// Tick data structure
struct TickData {
    double price;
    datetime time;
};

// Global variables
TickData tickPrices[];
int tickCount = 0;
int tickIndex = 0;
double bullishBodies1000[];
double bullishBodies250[];
double bearishBodies1000[];
double bearishBodies250[];
int bullBodies1000Count = 0;
int bullBodies250Count = 0;
int bearBodies1000Count = 0;
int bearBodies250Count = 0;
double bullCachedAvgBody1000 = 0.02000;
double bullCachedAvgBody250 = 0.00500;
double bearCachedAvgBody1000 = 0.02000;
double bearCachedAvgBody250 = 0.00500;
datetime lastStatsUpdate = 0;
double lastPrice = 0;

//+------------------------------------------------------------------+
//| Custom indicator initialization function                           |
//+------------------------------------------------------------------+
int init()
{
    // Set up indicator buffers
    IndicatorBuffers(4);
    SetIndexBuffer(0, BullTrendStrengthBuffer);
    SetIndexBuffer(1, BullRetracementBuffer);
    SetIndexBuffer(2, BearTrendStrengthBuffer);
    SetIndexBuffer(3, BearRetracementBuffer);
    
    // Set drawing properties
    SetIndexStyle(0, DRAW_LINE);
    SetIndexStyle(1, DRAW_LINE);
    SetIndexStyle(2, DRAW_LINE);
    SetIndexStyle(3, DRAW_LINE);
    
    // Set empty values
    SetIndexEmptyValue(0, EMPTY_VALUE);
    SetIndexEmptyValue(1, EMPTY_VALUE);
    SetIndexEmptyValue(2, EMPTY_VALUE);
    SetIndexEmptyValue(3, EMPTY_VALUE);
    
    // Set labels
    IndicatorShortName("Tick Trend Retrace Combined");
    SetIndexLabel(0, "Bull Trend");
    SetIndexLabel(1, "Bull Retrace");
    SetIndexLabel(2, "Bear Trend");
    SetIndexLabel(3, "Bear Retrace");
    
    // Initialize arrays
    ArrayResize(tickPrices, TickPerCandle);
    ArrayResize(bullishBodies1000, LookbackPeriod);
    ArrayResize(bullishBodies250, LookbackPeriod);
    ArrayResize(bearishBodies1000, LookbackPeriod);
    ArrayResize(bearishBodies250, LookbackPeriod);
    
    // Reset all counters
    tickCount = 0;
    bullBodies1000Count = bullBodies250Count = 0;
    bearBodies1000Count = bearBodies250Count = 0;
    lastStatsUpdate = 0;
    lastPrice = 0;
    
    // Clear buffers
    ArrayInitialize(BullTrendStrengthBuffer, EMPTY_VALUE);
    ArrayInitialize(BullRetracementBuffer, EMPTY_VALUE);
    ArrayInitialize(BearTrendStrengthBuffer, EMPTY_VALUE);
    ArrayInitialize(BearRetracementBuffer, EMPTY_VALUE);
    
    CreateStatsObjects();
    
    return(INIT_SUCCEEDED);
}

//+------------------------------------------------------------------+
//| Create stats objects                                               |
//+------------------------------------------------------------------+
void CreateStatsObjects()
{
    int yDistance = 20;
    for(int i = 0; i < 8; i++)
    {
        string objName = "StatsText" + IntegerToString(i);
        ObjectCreate(0, objName, OBJ_LABEL, 0, 0, 0);
        ObjectSetInteger(0, objName, OBJPROP_CORNER, CORNER_LEFT_UPPER);
        ObjectSetInteger(0, objName, OBJPROP_XDISTANCE, 10);
        ObjectSetInteger(0, objName, OBJPROP_YDISTANCE, yDistance);
        ObjectSetInteger(0, objName, OBJPROP_COLOR, i < 4 ? clrGreen : clrRed);
        yDistance += 20;
    }
}

//+------------------------------------------------------------------+
//| Custom indicator deinitialization function                         |
//+------------------------------------------------------------------+
int deinit()
{
    for(int i = 0; i < 8; i++)
    {
        ObjectDelete(0, "StatsText" + IntegerToString(i));
    }
    return(0);
}

//+------------------------------------------------------------------+
//| Add new tick data                                                 |
//+------------------------------------------------------------------+
void AddTick(double price, datetime time)
{
    if(MathAbs(price - lastPrice) < 0.000001) return;
    
    lastPrice = price;
    tickPrices[tickIndex].price = price;
    tickPrices[tickIndex].time = time;
    
    if(tickCount < TickPerCandle) tickCount++;
    tickIndex = (tickIndex + 1) % TickPerCandle;
}

//+------------------------------------------------------------------+
//| Add body to history                                               |
//+------------------------------------------------------------------+
void AddBody(double body, bool isLongPeriod, bool isBull)
{
    if(isBull)
    {
        if(isLongPeriod)
        {
            if(bullBodies1000Count < LookbackPeriod)
            {
                bullishBodies1000[bullBodies1000Count] = body;
                bullBodies1000Count++;
            }
            else
            {
                for(int i = 0; i < LookbackPeriod-1; i++)
                    bullishBodies1000[i] = bullishBodies1000[i+1];
                bullishBodies1000[LookbackPeriod-1] = body;
            }
        }
        else
        {
            if(bullBodies250Count < LookbackPeriod)
            {
                bullishBodies250[bullBodies250Count] = body;
                bullBodies250Count++;
            }
            else
            {
                for(int i = 0; i < LookbackPeriod-1; i++)
                    bullishBodies250[i] = bullishBodies250[i+1];
                bullishBodies250[LookbackPeriod-1] = body;
            }
        }
    }
    else
    {
        if(isLongPeriod)
        {
            if(bearBodies1000Count < LookbackPeriod)
            {
                bearishBodies1000[bearBodies1000Count] = body;
                bearBodies1000Count++;
            }
            else
            {
                for(int i = 0; i < LookbackPeriod-1; i++)
                    bearishBodies1000[i] = bearishBodies1000[i+1];
                bearishBodies1000[LookbackPeriod-1] = body;
            }
        }
        else
        {
            if(bearBodies250Count < LookbackPeriod)
            {
                bearishBodies250[bearBodies250Count] = body;
                bearBodies250Count++;
            }
            else
            {
                for(int i = 0; i < LookbackPeriod-1; i++)
                    bearishBodies250[i] = bearishBodies250[i+1];
                bearishBodies250[LookbackPeriod-1] = body;
            }
        }
    }
}

//+------------------------------------------------------------------+
//| Update Statistics                                                 |
//+------------------------------------------------------------------+
void UpdateStatistics()
{
    datetime currentTime = TimeCurrent();
    if(currentTime - lastStatsUpdate < 300) return;
    
    // Update bull stats
    double bullSum1000 = 0, bullSum250 = 0;
    for(int i = 0; i < bullBodies1000Count; i++)
        bullSum1000 += bullishBodies1000[i];
    for(int i = 0; i < bullBodies250Count; i++)
        bullSum250 += bullishBodies250[i];
    
    bullCachedAvgBody1000 = bullBodies1000Count > 0 ? bullSum1000 / bullBodies1000Count : 0.02000;
    bullCachedAvgBody250 = bullBodies250Count > 0 ? bullSum250 / bullBodies250Count : 0.00500;
    
    // Update bear stats
    double bearSum1000 = 0, bearSum250 = 0;
    for(int i = 0; i < bearBodies1000Count; i++)
        bearSum1000 += bearishBodies1000[i];
    for(int i = 0; i < bearBodies250Count; i++)
        bearSum250 += bearishBodies250[i];
    
    bearCachedAvgBody1000 = bearBodies1000Count > 0 ? bearSum1000 / bearBodies1000Count : 0.02000;
    bearCachedAvgBody250 = bearBodies250Count > 0 ? bearSum250 / bearBodies250Count : 0.00500;
    
    lastStatsUpdate = currentTime;
    
    // Update display
    ObjectSetString(0, "StatsText0", OBJPROP_TEXT, StringFormat("Bull AvgBody1000: %.8f", bullCachedAvgBody1000));
    ObjectSetString(0, "StatsText1", OBJPROP_TEXT, StringFormat("Bull AvgBody250: %.8f", bullCachedAvgBody250));
    ObjectSetString(0, "StatsText2", OBJPROP_TEXT, StringFormat("Bull Bodies1000: %d", bullBodies1000Count));
    ObjectSetString(0, "StatsText3", OBJPROP_TEXT, StringFormat("Bull Bodies250: %d", bullBodies250Count));
    ObjectSetString(0, "StatsText4", OBJPROP_TEXT, StringFormat("Bear AvgBody1000: %.8f", bearCachedAvgBody1000));
    ObjectSetString(0, "StatsText5", OBJPROP_TEXT, StringFormat("Bear AvgBody250: %.8f", bearCachedAvgBody250));
    ObjectSetString(0, "StatsText6", OBJPROP_TEXT, StringFormat("Bear Bodies1000: %d", bearBodies1000Count));
    ObjectSetString(0, "StatsText7", OBJPROP_TEXT, StringFormat("Bear Bodies250: %d", bearBodies250Count));
}

//+------------------------------------------------------------------+
//| Custom indicator iteration function                                |
//+------------------------------------------------------------------+
int start()
{
    double currentPrice = MarketInfo(Symbol(), MODE_BID);
    datetime currentTime = TimeCurrent();
    
    AddTick(currentPrice, currentTime);
    
    if(tickCount < TickPerCandle) return(0);
    
    int startIndex = (tickIndex - tickCount + TickPerCandle) % TickPerCandle;
    double openPrice1000 = tickPrices[startIndex].price;
    double body1000 = MathAbs(currentPrice - openPrice1000);
    
    // Calculate trends
    if(currentPrice > openPrice1000)
    {
        AddBody(body1000, true, true);
    }
    else if(currentPrice < openPrice1000)
    {
        AddBody(body1000, true, false);
    }
    
    // Calculate normalized trends
    double bullNormalizedTrend = (currentPrice - openPrice1000) / bullCachedAvgBody1000;
    double bearNormalizedTrend = (currentPrice - openPrice1000) / bearCachedAvgBody1000;
    
    BullTrendStrengthBuffer[0] = MathMax(-2, MathMin(2, bullNormalizedTrend));
    BearTrendStrengthBuffer[0] = MathMax(-2, MathMin(2, bearNormalizedTrend));
    
    // Calculate retracements if enough ticks
    if(tickCount >= RetracementUpdateInterval)
    {
        int retraceStartIdx = (tickIndex - RetracementUpdateInterval + TickPerCandle) % TickPerCandle;
        double openPrice250 = tickPrices[retraceStartIdx].price;
        double body250 = MathAbs(currentPrice - openPrice250);
        
        if(currentPrice > openPrice250)
        {
            AddBody(body250, false, true);
        }
        else if(currentPrice < openPrice250)
        {
            AddBody(body250, false, false);
        }
        
        // Calculate normalized retracements
        double bullNormalizedRetracement = (currentPrice - openPrice250) / bullCachedAvgBody250;
        double bearNormalizedRetracement = (currentPrice - openPrice250) / bearCachedAvgBody250;
        
        BullRetracementBuffer[0] = MathMax(-2, MathMin(2, bullNormalizedRetracement));
        BearRetracementBuffer[0] = MathMax(-2, MathMin(2, bearNormalizedRetracement));
    }
    
    UpdateStatistics();
    return(0);
}
