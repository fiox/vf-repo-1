//+------------------------------------------------------------------+
//| Main tick function                                               |
//+------------------------------------------------------------------+
void OnTick()
{
    // HLCO_60 logic (EA1)
    datetime currentTime = TimeCurrent();
    double currentPrice = Bid;
    AddTick(currentPrice, currentTime);
    RemoveOldTicks(currentTime);
    double high = CalculateHigh();
    double low = CalculateLow();
    double open = CalculateOpen();
    double close = CalculateClose();
    LastHigh = high;
    LastLow = low;
    LastOpen = open;
    LastClose = close;
    UpdateLinesAndLabels();
}

        }
    }
}

//+------------------------------------------------------------------+
//| Functions for HLCO_60 (EA1)                                      |
//+------------------------------------------------------------------+
void AddTick(double price, datetime time)
{
    int size = ArraySize(TickPrices);
    ArrayResize(TickPrices, size + 1);
    ArrayResize(TickTimes, size + 1);
    TickPrices[size] = price;
    TickTimes[size] = time;
}

void RemoveOldTicks(datetime currentTime)
{
    int newSize = 0;
    for (int i = 0; i < ArraySize(TickPrices); i++)
    {
        if (currentTime - TickTimes[i] <= TimeFrameSeconds)
        {
            TickPrices[newSize] = TickPrices[i];
            TickTimes[newSize] = TickTimes[i];
            newSize++;
        }
    }
    ArrayResize(TickPrices, newSize);
    ArrayResize(TickTimes, newSize);
}

double CalculateHigh()
{
    double high = EMPTY_VALUE;
    for (int i = 0; i < ArraySize(TickPrices); i++)
    {
        if (TickPrices[i] > high || high == EMPTY_VALUE)
        {
            high = TickPrices[i];
        }
    }
    return high;
}

double CalculateLow()
{
    double low = EMPTY_VALUE;
    for (int i = 0; i < ArraySize(TickPrices); i++)
    {
        if (TickPrices[i] < low || low == EMPTY_VALUE)
        {
            low = TickPrices[i];
        }
    }
    return low;
}

double CalculateOpen()
{
    if (ArraySize(TickPrices) > 0)
    {
        return TickPrices[0]; // The oldest tick in the array
    }
    return EMPTY_VALUE;
}

double CalculateClose()
{
    if (ArraySize(TickPrices) > 0)
    {
        return TickPrices[ArraySize(TickPrices) - 1]; // The most recent tick in the array
    }
    return EMPTY_VALUE;
}

    DrawLabel(LowLabelName, "Low", LastLow, clrBlue);
    DrawHorizontalLine(OpenLineName, LastOpen, clrGreen);
    DrawLabel(OpenLabelName, "Open", LastOpen, clrGreen);
    DrawHorizontalLine(CloseLineName, LastClose, clrDarkOrange);
    DrawLabel(CloseLabelName, "Close", LastClose, clrDarkOrange);
}

void DrawHorizontalLine(string name, double value, color lineColor)
{
    if (ObjectFind(0, name) < 0)
    {
        ObjectCreate(0, name, OBJ_HLINE, 0, 0, value);
    }
    ObjectSetInteger(0, name, OBJPROP_COLOR, lineColor);
    ObjectSetDouble(0, name, OBJPROP_PRICE, value);
    ObjectSetInteger(0, name, OBJPROP_WIDTH, 1);
    ObjectSetInteger(0, name, OBJPROP_STYLE, STYLE_SOLID);
}

void DrawLabel(string name, string text, double price, color textColor)
{
    if (ObjectFind(0, name) < 0)
    {
        ObjectCreate(0, name, OBJ_TEXT, 0, Time[0], price);
    }
    datetime labelTime = Time[0] + LabelOffsetBars * PeriodSeconds();
    ObjectSetString(0, name, OBJPROP_TEXT, text);
    ObjectSetInteger(0, name, OBJPROP_COLOR, textColor);
    ObjectSetDouble(0, name, OBJPROP_PRICE, price);
    ObjectSetInteger(0, name, OBJPROP_TIME, labelTime);
    ObjectSetInteger(0, name, OBJPROP_ANCHOR, ANCHOR_RIGHT_UPPER);
}

// OPTIMIZED Close_Buy_Button FUNCTION - Faster execution
int Close_Buy_Button(const string sparam)
{
    if (OrdersTotal() == 0) return(0);
    
    int closed = 0;
    
    for (int i = OrdersTotal() - 1; i >= 0; i--)
    {
        // Single OrderSelect with all conditions in one check for efficiency
        if (!OrderSelect(i, SELECT_BY_POS, MODE_TRADES) || OrderSymbol() != Symbol() || OrderType() != OP_BUY)
            continue;
        
        // Store order info to avoid multiple OrderSelect calls
        int orderTicket = OrderTicket();
        double orderLots = OrderLots();
        bool success = false;
        
        // Try with increasing slippage - minimize delay between retries
        for (int attempt = 1; attempt <= 3; attempt++)
        {
            int slippage = slip * attempt;
            
            // Reselect only once per attempt to get fresh price
            if (OrderSelect(orderTicket, SELECT_BY_TICKET))
            {
                if (OrderClose(orderTicket, orderLots, Bid, slippage, CLR_NONE))
                {
                    closed++;
                    success = true;
                    break;
                }
                else
                {
                    int error = GetLastError();
                    // Minimal wait on requote, increasing with each attempt
                    if (error == 138)
                        Sleep(20 * attempt);
                    else
                        Print("Error closing buy: ", error);
                }
            }
        }
        
        if (!success) Print("Failed to close buy position after 3 attempts");
    }
    
    if (closed > 0) Print("Closed ", closed, " buy positions");
    return(0);
}

// OPTIMIZED Close_All_Button FUNCTION - Faster execution with OrderCloseBy for matching positions
int Close_All_Button(const string sparam)
{
    if (OrdersTotal() == 0) return(0);
    
    // Stats tracking
    int closedPairs = 0;
    int closedBuys = 0;
    int closedSells = 0;
    
    // First try to use OrderCloseBy for matching positions (to avoid commissions)
    for (int b = OrdersTotal() - 1; b >= 0; b--)
    {
        if (!OrderSelect(b, SELECT_BY_POS, MODE_TRADES) || OrderSymbol() != Symbol() || OrderType() != OP_BUY)
            continue;
            
        double buyLot = OrderLots();
        int buyTicket = OrderTicket();
        
        // Look for matching sell order
        for (int s = OrdersTotal() - 1; s >= 0; s--)
        {
            if (!OrderSelect(s, SELECT_BY_POS, MODE_TRADES) || OrderSymbol() != Symbol() || OrderType() != OP_SELL)
                continue;
                
            if (NormalizeDouble(OrderLots(), 2) == NormalizeDouble(buyLot, 2))
            {
                // Found a matching sell order, try to close the pair
                int sellTicket = OrderTicket();
                
                // Reselect the buy ticket as current order
                if (OrderSelect(buyTicket, SELECT_BY_TICKET))
                {
                    if (OrderCloseBy(buyTicket, sellTicket, CLR_NONE))
                    {
                        closedPairs++;
                        // Since OrdersTotal() will now be different, we need to adjust
                        b = OrdersTotal(); // This will become OrdersTotal()-1 after the b-- in the loop
                        if (b < 0) break;  // No more orders
                        break;
                    }
                }
            }
        }
    }
    
    // Close remaining buys (that couldn't be paired)
    for (int i = OrdersTotal() - 1; i >= 0; i--)
    {
        if (!OrderSelect(i, SELECT_BY_POS, MODE_TRADES) || OrderSymbol() != Symbol() || OrderType() != OP_BUY)
            continue;
            
        int orderTicket = OrderTicket();
        double orderLots = OrderLots();
        bool success = false;
        
        // Try with increasing slippage
        for (int attempt = 1; attempt <= 3; attempt++)
        {
            int slippage = slip * attempt;
            
            if (OrderSelect(orderTicket, SELECT_BY_TICKET))
            {
                if (OrderClose(orderTicket, orderLots, Bid, slippage, CLR_NONE))
                {
                    closedBuys++;
                    success = true;
                    break;
                }
                else
                {
                    int error = GetLastError();
                    if (error == 138) // Requote - use minimal wait time
                        Sleep(20 * attempt);
                    else
                        Print("Buy close error: ", error);
                }
            }
        }
        
        if (!success) Print("Failed to close buy position after 3 attempts");
    }
    
    // Close remaining sells (that couldn't be paired)
    for (int j = OrdersTotal() - 1; j >= 0; j--)
    {
        if (!OrderSelect(j, SELECT_BY_POS, MODE_TRADES) || OrderSymbol() != Symbol() || OrderType() != OP_SELL)
            continue;
            
        int orderTicket2 = OrderTicket();
        double orderLots2 = OrderLots();
        bool success2 = false;
        
        // Try with increasing slippage
        for (int attempt2 = 1; attempt2 <= 3; attempt2++)
        {
            int slippage2 = slip * attempt2;
            
            if (OrderSelect(orderTicket2, SELECT_BY_TICKET))
            {
                if (OrderClose(orderTicket2, orderLots2, Ask, slippage2, CLR_NONE))
                {
                    closedSells++;
                    success2 = true;
                    break;
                }
                else
                {
                    int error2 = GetLastError();
                    if (error2 == 138) // Requote - use minimal wait time
                        Sleep(20 * attempt2);
                    else
                        Print("Sell close error: ", error2);
                }
            }
        }
        
        if (!success2) Print("Failed to close sell position after 3 attempts");
    }
    
    // Report results
    if (closedPairs > 0 || closedBuys > 0 || closedSells > 0)
        Print("Closed: ", closedPairs, " pairs, ", closedBuys, " buys, ", closedSells, " sells");
    
    return(0);
}
